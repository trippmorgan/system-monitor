def clean_title(element):
    if element is None or element.text is None:
        return 'Untitled'
    text = element.text
    text = re.sub(r'<[^>]+>', '', text)   # Strip HTML tags from CDATA
    text = html.unescape(text)             # Decode &amp; etc.
    text = ' '.join(text.split())          # Collapse whitespace
    return text.strip() or 'Untitled'
```

def get_link(item):
    link = item.find('link')
    if link is not None:
        url = link.text or link.get('href', '')
        if url and url.strip().startswith(('http://', 'https://')):
            return url.strip()
    return ''
```
This handles both RSS `<link>text</link>` and Atom `<link href="..."/>` formats, and validates URLs start with http(s) (XSS protection since dashboard uses innerHTML).
